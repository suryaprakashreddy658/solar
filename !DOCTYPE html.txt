    t_bill.innerText="నెలవారీ విద్యుత్ బిల్లు (₹)";
    t_units.innerText="లేదా యూనిట్లు (kWh)";
    t_sub.innerText="తెలంగాణ సౌర సబ్సిడీ";
  } else {
    location.reload();
  }
}

function calculate(){
  let billVal=parseFloat(bill.value);
  let unitsVal=parseFloat(units.value);

  if(!unitsVal && billVal) unitsVal=billVal/7;
  if(!unitsVal) return alert("Please enter bill or units");

  let kw=Math.round((unitsVal/130)*2)/2;
  let base=kw*50000;

  let subsidy = kw<=3
    ? base*0.4
    : (3*50000*0.4)+((kw-3)*50000*0.2);

  let net=base-subsidy;
  let saving=kw*130*7;
  let payback=net/(saving*12);

  result.innerHTML=`
    <b>Recommended System Size:</b> ${kw} kW<br><br>
    <b>Base Cost:</b> ₹${base.toLocaleString()}<br>
    <b>Government Subsidy:</b> ₹${subsidy.toLocaleString()}<br>
    <b>Net Payable Cost:</b> ₹${net.toLocaleString()}<br><br>
    <b>Estimated Monthly Savings:</b> ₹${saving.toLocaleString()}<br>
    <b>Payback Period:</b> ${payback.toFixed(1)} years
  `;
}
